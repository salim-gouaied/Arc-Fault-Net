\section*{Dédicace}

\vspace*{2.5cm}

\begin{flushright}
\begin{minipage}{0.72\textwidth}
\raggedleft\itshape

À l'âme de mes grands-pères et grands-mères,\\
que Dieu les accueille en Sa miséricorde.\\[1cm]

À mes chers parents,\\
pour leur amour, leur soutien indéfectible\\
et leurs sacrifices tout au long de mon parcours.\\[1cm]

À ma famille,\\
pour ses encouragements constants.\\[1cm]

À tous mes amis et collègues\\
qui ont contribué, de près ou de loin,\\
à l'aboutissement de ce travail.\\[1cm]

\upshape\normalsize Moslem MOHAMED

\end{minipage}
\end{flushright}
